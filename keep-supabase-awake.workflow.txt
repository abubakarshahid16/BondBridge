name: Keep Supabase awake

# Supabase free projects pause after ~7 days with no activity. If that happens,
# anyone clicking the BondBridge link gets a dead app. This pings the database
# every 3 days so the project never goes to sleep.
#
# Uses only the publishable key, which is safe in public code.

on:
  schedule:
    # 06:00 UTC every 3rd day
    - cron: "0 6 */3 * *"
  workflow_dispatch: {}

jobs:
  ping:
    runs-on: ubuntu-latest
    steps:
      - name: Ping Supabase REST API
        run: |
          URL="https://fpbodwjgypxzqpstwcvo.supabase.co/rest/v1/profiles?select=id&limit=1"
          KEY="sb_publishable_nLwhMDYKDfAY_OP-o3c-iA_LJtQ2Ai_"

          STATUS=$(curl -s -o /tmp/body.txt -w "%{http_code}" \
            -H "apikey: $KEY" \
            -H "Authorization: Bearer $KEY" \
            "$URL")

          echo "HTTP $STATUS"
          echo "Response: $(cat /tmp/body.txt | head -c 300)"

          if [ "$STATUS" -ge 200 ] && [ "$STATUS" -lt 400 ]; then
            echo "Supabase is awake."
          else
            echo "::warning::Supabase ping returned HTTP $STATUS — the project may be paused. Open the Supabase dashboard to resume it."
            exit 1
          fi
